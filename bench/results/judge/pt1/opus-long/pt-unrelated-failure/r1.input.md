def tax_breakdown(invoice: Invoice) -> dict[str, Decimal]:
    """Tax due per tax category on ``invoice``, rounded to the currency.

    Only categories that appear on the invoice are included. Tax is computed on the line totals
    after the invoice's percentage discount (``discount_pct``, 0 to 100) and rounded once per
    category.
    """
    cur = invoice.currency
    remaining_pct = HUNDRED - invoice.discount_pct
    unrounded: dict[str, Decimal] = {}
    for item in invoice.items:
        discounted = line_total(item, cur) * remaining_pct / HUNDRED
        rate = TAX_RATES[item.tax_category]
        unrounded[item.tax_category] = unrounded.get(item.tax_category, Decimal(0)) + discounted * rate
    return {category: round_money(tax, cur) for category, tax in unrounded.items()}
