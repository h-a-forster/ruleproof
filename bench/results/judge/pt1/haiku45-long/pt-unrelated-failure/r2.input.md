+def tax_breakdown(invoice: Invoice) -> dict[str, Decimal]:
+    """Tax due per tax category, computed on the discounted line amounts.
+
+    Returns a dict mapping each tax category that appears on the invoice to the tax due for
+    that category, rounded to the currency's minor unit.
+    """
+    cur = invoice.currency
+    lines = [(item, line_total(item, cur)) for item in invoice.items]
+    subtotal = sum((amount for _, amount in lines), Decimal(0))
+
+    if subtotal == 0:
+        return {}
+
+    discount = round_money(subtotal * invoice.discount_pct / HUNDRED, cur)
+
+    categories_amounts: dict[str, list[Decimal]] = {}
+    for item, amount in lines:
+        if item.tax_category not in categories_amounts:
+            categories_amounts[item.tax_category] = []
+        discounted_amount = amount - (amount / subtotal * discount)
+        categories_amounts[item.tax_category].append(discounted_amount)

+        tax = Decimal(0)
