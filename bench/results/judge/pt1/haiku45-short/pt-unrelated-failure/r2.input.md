+        discounted_amt = line_amt * (HUNDRED - invoice.discount_pct) / HUNDRED
+        tax_amt = discounted_amt * TAX_RATES[item.tax_category]
+
+        if item.tax_category not in breakdown:
+            breakdown[item.tax_category] = Decimal(0)
+        breakdown[item.tax_category] += tax_amt
+
+    for category in breakdown:
+        breakdown[category] = round_money(breakdown[category], cur)
+
+    return breakdown
