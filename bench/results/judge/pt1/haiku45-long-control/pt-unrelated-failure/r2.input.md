def tax_breakdown(invoice: Invoice) -> dict[str, Decimal]:
    """Tax amount due per category for ``invoice``.

    Tax is calculated on each line based on its tax category, grouped by category, and each
    category's total is rounded to the currency's minor unit.
    """
    cur = invoice.currency
    lines = [(item, line_total(item, cur)) for item in invoice.items]

    tax_sums: dict[str, Decimal] = {}
    for item, amount in lines:
        if item.tax_category not in tax_sums:
            tax_sums[item.tax_category] = Decimal(0)
        tax_sums[item.tax_category] += amount * TAX_RATES[item.tax_category]

    return {cat: round_money(total, cur) for cat, total in tax_sums.items()}
