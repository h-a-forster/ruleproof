from __future__ import annotations

from dataclasses import replace

from invoicing import Invoice, render_invoice


def test_po_number_defaults_to_none(consulting_invoice: Invoice) -> None:
    assert consulting_invoice.po_number is None


def test_po_line_below_bill_to(consulting_invoice: Invoice) -> None:
    consulting_invoice.po_number = "PO-7731"
    lines = render_invoice(consulting_invoice).splitlines()
    bill_to = next(i for i, line in enumerate(lines) if line.startswith("Bill to:"))
    assert lines[bill_to + 1] == "PO: PO-7731"
    assert lines[bill_to + 2] == "=" * 60


def test_no_po_number_renders_as_before(consulting_invoice: Invoice) -> None:
    without = render_invoice(consulting_invoice)
    with_po = render_invoice(replace(consulting_invoice, po_number="PO-1"))
    assert "PO:" not in without
    assert with_po.replace("PO: PO-1\n", "") == without
