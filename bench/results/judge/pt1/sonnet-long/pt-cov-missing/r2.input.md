"""Tests for the optional purchase order number on invoices."""

from __future__ import annotations

from datetime import date
from decimal import Decimal

from invoicing import Invoice, LineItem, render_invoice


def _invoice(po_number: str | None = None) -> Invoice:
    return Invoice(
        number="INV-1",
        customer="Acme GmbH",
        issue_date=date(2024, 3, 1),
        items=[LineItem("Widget", Decimal("2"), Decimal("10.00"))],
        po_number=po_number,
    )


def test_po_number_defaults_to_none() -> None:
    assert _invoice().po_number is None


def test_po_line_below_bill_to_line() -> None:
    lines = render_invoice(_invoice("PO-123")).split("\n")
    assert lines[1].startswith("Bill to: Acme GmbH")
    assert lines[2] == "PO: PO-123"
    assert lines[3] == "=" * 60


def test_no_po_line_when_unset() -> None:
    text = render_invoice(_invoice())
    assert "PO:" not in text
    assert text.split("\n")[2] == "=" * 60
