def tax_breakdown(invoice: Invoice) -> dict[str, Decimal]:
    """Tax due per tax category on ``invoice``, in the invoice's currency.

    Tax is computed on each rounded line total after the invoice discount (``discount_pct``,
    0 to 100) is applied, summed per category and rounded once per category. Only categories
    that appear on the invoice are included.
    """
    cur = invoice.currency
    discount_factor = (HUNDRED - invoice.discount_pct) / HUNDRED
    unrounded: dict[str, Decimal] = {}
    for item in invoice.items:
        discounted = line_total(item, cur) * discount_factor
        category = item.tax_category
        unrounded[category] = unrounded.get(category, Decimal(0)) + discounted * TAX_RATES[category]
    return {category: round_money(tax, cur) for category, tax in unrounded.items()}
