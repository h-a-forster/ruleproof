def tax_breakdown(invoice: Invoice) -> dict[str, Decimal]:
    """Tax due per tax category on ``invoice``, in the invoice currency.

    Tax is computed on the discounted amount of each rounded line (``discount_pct`` is on a 0 to
    100 scale) and rounded once per category. Only categories present on the invoice appear.
    """
    cur = invoice.currency
    keep = (HUNDRED - invoice.discount_pct) / HUNDRED
    unrounded: dict[str, Decimal] = {}
    for item in invoice.items:
        category = item.tax_category
        discounted = line_total(item, cur) * keep
        unrounded[category] = unrounded.get(category, Decimal(0)) + discounted * TAX_RATES[category]
    return {category: round_money(tax, cur) for category, tax in unrounded.items()}
