"""Tests for line item description validation."""

from __future__ import annotations

from decimal import Decimal

import pytest

from invoicing import LineItem


def test_empty_description_rejected() -> None:
    with pytest.raises(ValueError, match="description must not be blank"):
        LineItem("", Decimal("1"), Decimal("10.00"))


def test_whitespace_description_rejected() -> None:
    with pytest.raises(ValueError, match="description must not be blank"):
        LineItem("  \t\n", Decimal("1"), Decimal("10.00"))


def test_normal_description_accepted() -> None:
    item = LineItem("Consulting", Decimal("1"), Decimal("10.00"))
    assert item.description == "Consulting"
